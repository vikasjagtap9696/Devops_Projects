============================================================
STEP 26 — CREATE GITHUB ACTIONS WORKFLOW
============================================================

File:

.github/workflows/ci-cd.yml


Use the following complete workflow:

name: Heart Disease MLOps CI/CD

on:
  push:
    branches:
      - main

jobs:

  build-and-push:
    name: Build and Push Docker Images
    runs-on: [self-hosted, Linux, X64]

    permissions:
      contents: read

    env:
      AWS_REGION: ap-south-1
      ECR_REPOSITORY: heart-disease-prediction-mlops

    steps:

      # ------------------------------------------------
      # 1. Checkout source code
      # ------------------------------------------------
      - name: Checkout code
        uses: actions/checkout@v4


      # ------------------------------------------------
      # 2. Verify AWS IAM Role
      # ------------------------------------------------
      - name: Verify AWS Identity
        run: |
          aws sts get-caller-identity


      # ------------------------------------------------
      # 3. Login to Amazon ECR
      # ------------------------------------------------
      - name: Login to Amazon ECR
        id: login-ecr
        uses: aws-actions/amazon-ecr-login@v2


      # ------------------------------------------------
      # 4. Build FastAPI Docker Image
      # ------------------------------------------------
      - name: Build FastAPI image
        run: |
          docker build \
            -f API/Dockerfile \
            -t ${{ steps.login-ecr.outputs.registry }}/${{ env.ECR_REPOSITORY }}:fastapi .


      # ------------------------------------------------
      # 5. Build MLflow Docker Image
      # ------------------------------------------------
      - name: Build MLflow image
        run: |
          docker build \
            -f Dockerfile \
            -t ${{ steps.login-ecr.outputs.registry }}/${{ env.ECR_REPOSITORY }}:mlflow .


      # ------------------------------------------------
      # 6. Build Training Docker Image
      # ------------------------------------------------
      - name: Build Training image
        run: |
          docker build \
            -f scripts/Dockerfile \
            -t ${{ steps.login-ecr.outputs.registry }}/${{ env.ECR_REPOSITORY }}:training .


      # ------------------------------------------------
      # 7. Push FastAPI Image
      # ------------------------------------------------
      - name: Push FastAPI image
        run: |
          docker push \
            ${{ steps.login-ecr.outputs.registry }}/${{ env.ECR_REPOSITORY }}:fastapi


      # ------------------------------------------------
      # 8. Push MLflow Image
      # ------------------------------------------------
      - name: Push MLflow image
        run: |
          docker push \
            ${{ steps.login-ecr.outputs.registry }}/${{ env.ECR_REPOSITORY }}:mlflow


      # ------------------------------------------------
      # 9. Push Training Image
      # ------------------------------------------------
      - name: Push Training image
        run: |
          docker push \
            ${{ steps.login-ecr.outputs.registry }}/${{ env.ECR_REPOSITORY }}:training


      # ------------------------------------------------
      # 10. Verify ECR Images
      # ------------------------------------------------
      - name: Verify ECR images
        run: |
          aws ecr describe-images \
            --repository-name $ECR_REPOSITORY \
            --region $AWS_REGION \
            --query 'imageDetails[*].imageTags' \
            --output table


      # ------------------------------------------------
      # 11. Deploy Application
      # ------------------------------------------------
      - name: Deploy application
        run: |
          cd /home/ec2-user/heart-disease-prediction-mlops

          export ECR_URI=${{ steps.login-ecr.outputs.registry }}

          echo "ECR Registry: $ECR_URI"

          docker-compose -f Docker-compose.prod.yaml pull

          docker-compose -f Docker-compose.prod.yaml up -d --force-recreate


      # ------------------------------------------------
      # 12. Verify Deployment
      # ------------------------------------------------
      - name: Verify application health
        run: |
          sleep 10

          curl --fail http://localhost:8000/health

          echo "FastAPI deployment successful"
